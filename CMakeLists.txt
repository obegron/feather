cmake_minimum_required(VERSION 3.22)
project(feather-plasma VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui DBus Qml Test Widgets)
find_package(KF6Config REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6JobWidgets REQUIRED)
find_package(KF6Notifications REQUIRED)
find_package(KF6Purpose REQUIRED)
add_library(feather-radio STATIC src/radioservice.cpp)
target_link_libraries(feather-radio PUBLIC Qt6::Core Qt6::DBus KF6::ConfigCore)
target_include_directories(feather-radio PUBLIC src)
add_executable(feather-session src/main.cpp src/transfermanager.cpp src/transfermanager.h)
target_link_libraries(feather-session PRIVATE feather-radio Qt6::Widgets KF6::KIOWidgets KF6::JobWidgets KF6::Notifications)
add_executable(feather-send src/send.cpp)
target_link_libraries(feather-send PRIVATE Qt6::Core Qt6::DBus)
target_compile_definitions(feather-session PRIVATE FEATHER_HELPER_PATH="${CMAKE_INSTALL_FULL_LIBDIR}/feather/feather-transfer")
find_program(CARGO_EXECUTABLE cargo REQUIRED)
file(GLOB_RECURSE FEATHER_RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/adapter/src/*.rs" "${CMAKE_CURRENT_SOURCE_DIR}/adapter/vendor/*.rs")
set(FEATHER_RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/rust-target")
add_custom_command(OUTPUT "${FEATHER_RUST_TARGET}/release/feather-transfer"
 COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${FEATHER_RUST_TARGET} "RUSTFLAGS=$ENV{RUSTFLAGS}" ${CARGO_EXECUTABLE} build --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/adapter/Cargo.toml --locked --release --offline
 DEPENDS ${FEATHER_RUST_SOURCES} adapter/Cargo.toml adapter/Cargo.lock adapter/vendor/owly-airdrop/Cargo.toml
 WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
 VERBATIM)
add_custom_target(feather-transfer ALL DEPENDS "${FEATHER_RUST_TARGET}/release/feather-transfer")
install(PROGRAMS "${FEATHER_RUST_TARGET}/release/feather-transfer" DESTINATION ${CMAKE_INSTALL_LIBDIR}/feather)
install(TARGETS feather-send RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
add_library(featherpurposeplugin MODULE src/purposeplugin.cpp src/purposeplugin.json)
target_link_libraries(featherpurposeplugin PRIVATE KF6::Purpose Qt6::DBus)
set_target_properties(featherpurposeplugin PROPERTIES PREFIX "" OUTPUT_NAME "io.github.obegron.feather" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins/kf6/purpose")
install(TARGETS featherpurposeplugin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kf6/purpose)
install(FILES data/io.github.obegron.feather.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/feather.notifyrc DESTINATION ${CMAKE_INSTALL_DATADIR}/knotifications6)
install(DIRECTORY adapter/THIRD_PARTY_LICENSES/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/feather-plasma/rust)
install(FILES adapter/vendor/owly-airdrop/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/feather-plasma RENAME Owly-LICENSE)
qt_policy(SET QTP0001 NEW)
qt_add_qml_module(featherbridge OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/io/github/obegron/feather URI io.github.obegron.feather VERSION 1.0 SOURCES src/bridge.h src/bridge.cpp)
target_link_libraries(featherbridge PRIVATE Qt6::DBus Qt6::Gui)
target_include_directories(featherbridge PRIVATE src)
set_target_properties(featherbridgeplugin PROPERTIES INSTALL_RPATH "$ORIGIN")
set(FEATHER_QML_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/qml" CACHE PATH "Qt QML import root")
install(TARGETS feather-session RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS featherbridge featherbridgeplugin LIBRARY DESTINATION ${FEATHER_QML_INSTALL_DIR}/io/github/obegron/feather)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io/github/obegron/feather/qmldir ${CMAKE_CURRENT_BINARY_DIR}/io/github/obegron/feather/featherbridge.qmltypes DESTINATION ${FEATHER_QML_INSTALL_DIR}/io/github/obegron/feather)
install(DIRECTORY applet/ DESTINATION ${CMAKE_INSTALL_DATADIR}/plasma/plasmoids/io.github.obegron.feather)
configure_file(data/io.github.obegron.feather.service.in io.github.obegron.feather.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io.github.obegron.feather.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
include(CTest)
if(BUILD_TESTING)
 add_executable(radio-test tests/radio-test.cpp src/bridge.cpp src/bridge.h)
 target_link_libraries(radio-test PRIVATE feather-radio Qt6::Test Qt6::Qml Qt6::Gui)
 add_test(NAME radio-lifecycle COMMAND dbus-run-session -- $<TARGET_FILE:radio-test>)
 add_executable(transfer-test tests/transfer-test.cpp src/transfermanager.cpp src/transfermanager.h)
 target_link_libraries(transfer-test PRIVATE feather-radio Qt6::Widgets Qt6::Test KF6::KIOWidgets KF6::JobWidgets KF6::Notifications)
 target_compile_definitions(transfer-test PRIVATE FEATHER_HELPER_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/fake-transfer.py")
 add_test(NAME transfer-pipeline COMMAND dbus-run-session -- $<TARGET_FILE:transfer-test>)
 set_tests_properties(transfer-pipeline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/transfer-test-config")
 add_executable(clipboard-test tests/clipboard-test.cpp src/bridge.cpp src/bridge.h)
 target_link_libraries(clipboard-test PRIVATE Qt6::Gui Qt6::Test Qt6::Qml Qt6::DBus)
 target_include_directories(clipboard-test PRIVATE src)
 add_test(NAME clipboard-files COMMAND dbus-run-session -- $<TARGET_FILE:clipboard-test>)
 set_tests_properties(clipboard-files PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_executable(purpose-test tests/purpose-test.cpp)
 target_link_libraries(purpose-test PRIVATE KF6::Purpose Qt6::DBus Qt6::Test)
 target_compile_definitions(purpose-test PRIVATE FEATHER_PLUGIN_ROOT="${CMAKE_CURRENT_BINARY_DIR}/plugins")
 add_dependencies(purpose-test featherpurposeplugin)
 add_test(NAME purpose-share COMMAND dbus-run-session -- $<TARGET_FILE:purpose-test>)
 set_tests_properties(purpose-share PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/purpose-test-config")
 add_test(NAME adapter-protocol COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${FEATHER_RUST_TARGET} ${CARGO_EXECUTABLE} test --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/adapter/Cargo.toml --offline --locked)
 add_test(NAME airdrop-library COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${FEATHER_RUST_TARGET} ${CARGO_EXECUTABLE} test --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/adapter/vendor/owly-airdrop/Cargo.toml --offline --features mdns --lib)
 set_tests_properties(radio-lifecycle PROPERTIES ENVIRONMENT "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()
