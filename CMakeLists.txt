cmake_minimum_required(VERSION 3.22)
project(feather-plasma VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core DBus Qml Test)
find_package(KF6Config REQUIRED)
add_library(feather-radio STATIC src/radioservice.cpp)
target_link_libraries(feather-radio PUBLIC Qt6::Core Qt6::DBus KF6::ConfigCore)
target_include_directories(feather-radio PUBLIC src)
add_executable(feather-session src/main.cpp)
target_link_libraries(feather-session PRIVATE feather-radio)
qt_policy(SET QTP0001 NEW)
qt_add_qml_module(featherbridge OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/org/kde/feather URI org.kde.feather VERSION 1.0 SOURCES src/bridge.h src/bridge.cpp)
target_link_libraries(featherbridge PRIVATE Qt6::DBus)
target_include_directories(featherbridge PRIVATE src)
set_target_properties(featherbridgeplugin PROPERTIES INSTALL_RPATH "$ORIGIN")
set(FEATHER_QML_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/qml" CACHE PATH "Qt QML import root")
install(TARGETS feather-session RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS featherbridge featherbridgeplugin LIBRARY DESTINATION ${FEATHER_QML_INSTALL_DIR}/org/kde/feather)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org/kde/feather/qmldir ${CMAKE_CURRENT_BINARY_DIR}/org/kde/feather/featherbridge.qmltypes DESTINATION ${FEATHER_QML_INSTALL_DIR}/org/kde/feather)
install(DIRECTORY applet/ DESTINATION ${CMAKE_INSTALL_DATADIR}/plasma/plasmoids/org.kde.feather)
configure_file(data/org.kde.Feather.service.in org.kde.Feather.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.kde.Feather.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
include(CTest)
if(BUILD_TESTING)
 add_executable(radio-test tests/radio-test.cpp src/bridge.cpp src/bridge.h)
 target_link_libraries(radio-test PRIVATE feather-radio Qt6::Test Qt6::Qml)
 add_test(NAME radio-lifecycle COMMAND dbus-run-session -- $<TARGET_FILE:radio-test>)
 set_tests_properties(radio-lifecycle PROPERTIES ENVIRONMENT "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()
